# bowl_settings.txt -- settings for bowl_resonator.py (keep it next to the script)
# Edit, save, then apply with:   sudo systemctl restart plantbowl
# Anything after # is a comment. Delete a line to go back to its default.

# --- how the plant drives the sound ------------------------------------------
sensitivity = 1.0      # how strongly touches swell the bowl and how small a touch can strike (1 ... 3).
                       # Safe to raise: thresholds never go below the plant's own resting noise, so
                       # higher values don't add hum or strikes when nobody touches the plant
smoothing = 2.0        # seconds of smoothing on the slow tone changes: higher = steadier

# --- the bowl -----------------------------------------------------------------
hum = 0.1              # resting level of the bowl's hum: 0 = silent until touched ... 0.3 = steady drone
brightness = 0.40      # 0 = dark and warm ... 1 = bright and shimmery (overtone balance)
resonance = 20         # seconds the main bowl rings after being struck (8 short ... 40 very long)
root = 50              # pitch of preset 1 and the octave of presets 2-6, as a MIDI note:
                       # 50 = D3, 55 = G3, 57 = A3 (easier on small speakers), 60 = C4
pitch_drift = 6        # cents of gentle pitch movement with the plant (0 = perfectly fixed pitch)

# --- percussion -----------------------------------------------------------------
strike_density = 0.35  # 0 = no mallet strikes ... 1 = most frequent; also sets how closely strikes
                       # can follow each other while the plant is being handled
wood = 0.4             # level of the rare muted wooden touches (0 = none)

# --- space and level ----------------------------------------------------------
reverb = 0.25          # 0 = dry ... 0.6 = very spacious
volume = 0             # dB: +6 or +10 if too quiet, -6 if too loud (a limiter prevents clipping)
